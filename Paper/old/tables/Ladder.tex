% \hshade{value} : paints the cell by accuracy; \pgfmathtruncatemacro forces the INTEGER xcolor's ! mixing needs.
\newcommand{\hshade}[1]{%
  \pgfmathtruncatemacro{\shadeval}{max(0,min(60,(#1-14)/(74-14)*65))}%
  \edef\tmpcol{accteal!\shadeval}%
  \expandafter\cellcolor\expandafter{\tmpcol}%
}
\newcommand{\hcell}[1]{\hshade{#1}#1}
\newcommand{\hbest}[1]{\hshade{#1}\textbf{#1}}
% \dcell{magnitude} : warm wash deepens with the text channel's recovery over V; prints the signed value.
\newcommand{\dcell}[1]{%
  \pgfmathtruncatemacro{\shadeval}{max(0,min(55,(#1-2)/(13-2)*55))}%
  \edef\tmpcol{recwarm!\shadeval}%
  \expandafter\cellcolor\expandafter{\tmpcol}$-$#1%
}
